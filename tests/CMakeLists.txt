cmake_minimum_required(VERSION 3.15)

# Also usable with cmake -S tests, without configuring Vulkan / GLFW / Assimp.
if(CMAKE_SOURCE_DIR STREQUAL CMAKE_CURRENT_SOURCE_DIR)
  project(Engine0MeshTests LANGUAGES CXX)
  include(CTest)
endif()

set(ENGINE0_GLM_INCLUDE_DIR "" CACHE PATH "Directory containing glm/vec3.hpp (optional alternative to glm::glm)")
if(NOT TARGET glm::glm)
  if(ENGINE0_GLM_INCLUDE_DIR)
    if(NOT EXISTS "${ENGINE0_GLM_INCLUDE_DIR}/glm/vec3.hpp")
      message(FATAL_ERROR "ENGINE0_GLM_INCLUDE_DIR must contain glm/vec3.hpp")
    endif()
    add_library(Engine0MeshGlm INTERFACE)
    target_include_directories(Engine0MeshGlm INTERFACE "${ENGINE0_GLM_INCLUDE_DIR}")
    add_library(glm::glm ALIAS Engine0MeshGlm)
  else()
    find_package(glm CONFIG REQUIRED)
  endif()
endif()

add_library(Engine0MeshCpu STATIC ../Engine_0/Mesh.cpp ../Engine_0/MeshDrawData.cpp)
target_include_directories(Engine0MeshCpu PUBLIC "${CMAKE_CURRENT_SOURCE_DIR}/../Engine_0")
target_link_libraries(Engine0MeshCpu PUBLIC glm::glm)
target_compile_features(Engine0MeshCpu PUBLIC cxx_std_20)

add_executable(Engine0MeshTests MeshTests.cpp)
add_executable(Engine0MeshDrawTests MeshDrawDataTests.cpp)
add_executable(Engine0MeshInspect ../Engine_0/examples/mesh_inspect/main.cpp)
foreach(target Engine0MeshTests Engine0MeshDrawTests Engine0MeshInspect)
  target_link_libraries(${target} PRIVATE Engine0MeshCpu)
  if(MINGW)
    # Keep CPU tools runnable from CTest / Explorer without MinGW runtime DLLs on PATH.
    target_link_options(${target} PRIVATE -static)
  endif()
endforeach()
foreach(target Engine0MeshCpu Engine0MeshTests Engine0MeshDrawTests Engine0MeshInspect)
  if(MSVC)
    target_compile_options(${target} PRIVATE /utf-8 /W4 /WX /EHsc)
  else()
    target_compile_options(${target} PRIVATE -Wall -Wextra -Wpedantic -Werror)
  endif()
endforeach()

if(BUILD_TESTING)
  add_test(NAME mesh.cpu COMMAND Engine0MeshTests)
  add_test(NAME mesh.draw_mapping COMMAND Engine0MeshDrawTests)
  add_test(NAME mesh.inspect COMMAND Engine0MeshInspect)
  set_tests_properties(mesh.inspect PROPERTIES PASS_REGULAR_EXPRESSION "Validate=PASS")
endif()
